1. Integrate mineflayer-pvp for combat capabilities:
```javascript
function setupCombat() {
  const pvp = require('mineflayer-pvp').plugin;
  this.bot.loadPlugin(pvp);
  
  // Configure combat settings
  this.bot.pvp.attackRange = 3;
  this.bot.pvp.followRange = 6;
  
  // Add combat methods
  this.attackEntity = (entityName) => {
    const entity = this.bot.entities[entityName];
    if (!entity) return false;
    
    this.bot.pvp.attack(entity);
    return true;
  };
  
  this.protectPlayer = (playerName) => {
    const player = this.bot.players[playerName]?.entity;
    if (!player) return false;
    
    // Start protection mode
    this.protectionTarget = player;
    this.isProtecting = true;
    this.startProtectionLoop();
    return true;
  };
}
```
