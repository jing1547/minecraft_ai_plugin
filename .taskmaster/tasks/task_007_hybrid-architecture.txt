```java
public class AIConversationService {
    private final OpenAiService openAiService;
    private final Map<UUID, List<ChatMessage>> conversationHistory;
    private final Map<String, String> characterPersonalities;
    
    // Methods for processing conversations
    public String generateResponse(UUID playerId, String input, String character) {
        List<ChatMessage> history = getOrCreateHistory(playerId);
        String systemPrompt = characterPersonalities.get(character);
        
        // Add system message for character personality
        if (history.isEmpty()) {
            history.add(new ChatMessage("system", systemPrompt));
        }
        
        // Add user message
        history.add(new ChatMessage("user", input));
        
        // Generate response using GPT-4
        ChatCompletionRequest request = ChatCompletionRequest.builder()
            .model("gpt-4")
            .messages(history)
            .temperature(0.7)
            .maxTokens(300)
            .build();
            
        ChatMessage response = openAiService.createChatCompletion(request)
            .getChoices().get(0).getMessage();
            
        // Add response to history and return
        history.add(response);
        manageHistorySize(history);
        return response.getContent();
    }
}
```
