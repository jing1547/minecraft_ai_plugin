    try {
      await this.bot.builder.build(schematic, position);
      return true;
    } catch (err) {
      console.error('Building failed:', err);
      return false;
    }
  };
  
  this.collectRequiredMaterials = async (schematic) => {
    const materials = this.bot.builder.getMaterialsNeeded(schematic);

}
